Создай файл:

experiments/chatgpt_plus_probe.py
Python
from assistant.cdp import ChromeCDP

cdp = ChromeCDP()
cdp.connect()

page = cdp.find_page("chatgpt.com")

print("CONNECTED")

page.locator("#composer-plus-btn").click()

print("PLUS CLICKED")

html = page.content()

with open(
    "chatgpt_after_plus.html",
    "w",
    encoding="utf-8"
) as f:
    f.write(html)

print("HTML SAVED")

input("ENTER...")
